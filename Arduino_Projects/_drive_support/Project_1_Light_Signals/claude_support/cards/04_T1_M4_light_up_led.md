// Blinks the external LED wired to digital pin 9.
const int LED_PIN = 9;
void setup() {
pinMode(LED_PIN, OUTPUT);
}
void loop() {
digitalWrite(LED_PIN, HIGH);
delay(500);
digitalWrite(LED_PIN, LOW);
delay(500);
}

מנסים את אלה לפי הסדר:
1בודקים את כיוון הלד. הרגל הארוכה (+) דרך הנגד לחיבור דיגיטלי 9 של הארדואינו, הקצרה (−) ל־GND. אם הפוך — הלד לא יאיר. הופכים אותו.
2בודקים את הנגד. חייב להיות בטור — בין חיבור דיגיטלי 9 של הארדואינו לרגל הארוכה — לא סתם תקוע בברדבורד.
